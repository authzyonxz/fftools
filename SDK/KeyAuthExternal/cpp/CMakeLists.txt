cmake_minimum_required(VERSION 3.16)
project(KeyAuthTransport LANGUAGES CXX)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(keyauth_transport STATIC Transport.cpp)
target_compile_features(keyauth_transport PUBLIC cxx_std_17)
target_link_libraries(keyauth_transport PUBLIC OpenSSL::Crypto)
target_include_directories(keyauth_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(transport_cli ../tests/transport_cli.cpp)
target_link_libraries(transport_cli PRIVATE keyauth_transport)
